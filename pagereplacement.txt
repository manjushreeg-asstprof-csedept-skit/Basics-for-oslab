#include <stdio.h>

int findLRU(int time[], int n) {
    int i, minimum = time[0], pos = 0;
    for(i = 1; i < n; i++) {
        if(time[i] < minimum) {
            minimum = time[i];
            pos = i;
        }
    }
    return pos;
}

int main() {
    int framesCount, pagesCount, frames[10], pages[30];
    int faults = 0, hits = 0;
    int i, j, k, flag, pos;
    int counter = 0, time[10];

    printf("Enter number of frames: ");
    scanf("%d", &framesCount);

    printf("Enter number of pages: ");
    scanf("%d", &pagesCount);

    printf("Enter page reference string: ");
    for(i = 0; i < pagesCount; i++)
        scanf("%d", &pages[i]);

    // FIFO Page Replacement
    faults = hits = 0;
    for(i = 0; i < framesCount; i++)
        frames[i] = -1;

    j = 0;
    printf("\n--- FIFO Page Replacement ---\n");
    for(i = 0; i < pagesCount; i++) {
        flag = 0;
        for(k = 0; k < framesCount; k++) {
            if(frames[k] == pages[i]) {
                flag = 1;
                hits++;
                break;
            }
        }
        if(flag == 0) {
            frames[j] = pages[i];
            j = (j + 1) % framesCount;
            faults++;
        }

        printf("Frames: ");
        for(k = 0; k < framesCount; k++)
            printf("%d ", frames[k]);
        printf("\n");
    }

    printf("FIFO Total Page Faults = %d\n", faults);

    // LRU Page Replacement
    faults = hits = 0;
    for(i = 0; i < framesCount; i++) {
        frames[i] = -1;
    }

    printf("\n--- LRU Page Replacement ---\n");
    for(i = 0; i < pagesCount; i++) {
        flag = 0;

        for(j = 0; j < framesCount; j++) {
            if(frames[j] == pages[i]) {
                flag = 1;
                hits++;
                counter++;
                time[j] = counter;
                break;
            }
        }

        if(flag == 0) {
            for(j = 0; j < framesCount; j++) {
                if(frames[j] == -1) {
                    counter++;
                    faults++;
                    frames[j] = pages[i];
                    time[j] = counter;
                    flag = 1;
                    break;
                }
            }
        }

        if(flag == 0) {
            pos = findLRU(time, framesCount);
            counter++;
            faults++;
            frames[pos] = pages[i];
            time[pos] = counter;
        }

        printf("Frames: ");
        for(j = 0; j < framesCount; j++)
            printf("%d ", frames[j]);
        printf("\n");
    }

    printf("LRU Total Page Faults = %d\n", faults);

    return 0;
}
